// prisma/seed.ts
import type { User, Channel, DirectConversation, Message, Workspace } from '@prisma/client'

import { hash } from 'bcrypt'
import { PrismaClient } from '@prisma/client'

import { encryptAES } from '../src/common/utils'

const prisma = new PrismaClient()

async function main(): Promise<void> {
  try {
    console.log('Starting database seeding...')

    // Clean up any existing data if needed
    console.log('Cleaning up existing data...')
    await cleanup()

    // Create users
    console.log('Creating users...')
    const users = await createUsers()

    // Create workspaces
    console.log('Creating workspaces...')
    const workspaces = await createWorkspaces(users)

    // Add users to workspaces
    console.log('Adding users to workspaces...')
    await addUsersToWorkspaces(users, workspaces)

    // Create channels
    console.log('Creating channels...')
    const channels = await createChannels(workspaces)

    // Add users to channels
    console.log('Adding users to channels...')
    await addUsersToChannels(users, channels)

    // Create direct conversations
    console.log('Creating direct conversations...')
    const directConversations = await createDirectConversations(users)

    // Create messages in channels
    console.log('Creating channel messages...')
    await createChannelMessages(users, channels)

    // Create messages in direct conversations
    console.log('Creating direct messages...')
    await createDirectMessages(users, directConversations)

    // Update message statuses
    console.log('Updating message statuses...')
    await updateMessageStatuses()

    console.log('Database seeding completed successfully!')
  } catch (error) {
    console.error('Error seeding database:', error)
    throw error
  } finally {
    await prisma.$disconnect()
  }
}

async function cleanup(): Promise<void> {
  // Delete all existing data in reverse order of dependencies
  await prisma.reaction.deleteMany({})
  await prisma.attachment.deleteMany({})
  await prisma.directMessageStatus.deleteMany({})
  await prisma.message.deleteMany({})
  await prisma.directConversationMember.deleteMany({})
  await prisma.channelMember.deleteMany({})
  await prisma.directConversation.deleteMany({})
  await prisma.channel.deleteMany({})
  await prisma.workspaceMember.deleteMany({})
  await prisma.workspace.deleteMany({})
  await prisma.user.deleteMany({})
}

async function createWorkspaces(users: User[]): Promise<Workspace[]> {
  const workspaces = await Promise.all([
    prisma.workspace.create({
      data: {
        name: 'Main Workspace',
        description: 'The primary workspace for the team',
        creatorId: users[0].id,
      },
    }),
    prisma.workspace.create({
      data: {
        name: 'Project Alpha',
        description: 'Workspace for Project Alpha development',
        creatorId: users[1].id,
      },
    }),
  ])

  console.log(`Created ${workspaces.length} workspaces`)
  return workspaces
}

async function addUsersToWorkspaces(users: User[], workspaces: Workspace[]): Promise<void> {
  const workspaceMembers = []

  // Add all users to main workspace (first workspace)
  for (const user of users) {
    // Skip creator as they're already a member
    if (user.id === workspaces[0].creatorId) {
      workspaceMembers.push(
        await prisma.workspaceMember.create({
          data: {
            userId: user.id,
            workspaceId: workspaces[0].id,
            role: 'OWNER', // Creator is OWNER
          },
        })
      )
    } else {
      workspaceMembers.push(
        await prisma.workspaceMember.create({
          data: {
            userId: user.id,
            workspaceId: workspaces[0].id,
            role: 'MEMBER',
          },
        })
      )
    }
  }

  // Add first 3 users to Project Alpha workspace (second workspace)
  for (let i = 0; i < 3; i++) {
    // Skip creator as they're already a member
    if (users[i].id === workspaces[1].creatorId) {
      workspaceMembers.push(
        await prisma.workspaceMember.create({
          data: {
            userId: users[i].id,
            workspaceId: workspaces[1].id,
            role: 'OWNER', // Creator is OWNER
          },
        })
      )
    } else {
      const role = i === 0 ? 'ADMIN' : 'MEMBER'
      workspaceMembers.push(
        await prisma.workspaceMember.create({
          data: {
            userId: users[i].id,
            workspaceId: workspaces[1].id,
            role: role,
          },
        })
      )
    }
  }

  console.log(`Added ${workspaceMembers.length} workspace memberships`)
}

async function createChannels(workspaces: Workspace[]): Promise<Channel[]> {
  const channels = await Promise.all([
    // Main workspace channels
    prisma.channel.create({
      data: {
        name: 'general',
        description: 'General discussion for everyone',
        isPrivate: false,
        memberCount: 0, // Will be updated after adding members
        workspaceId: workspaces[0].id,
      },
    }),
    prisma.channel.create({
      data: {
        name: 'random',
        description: 'Random chit-chat and memes',
        isPrivate: false,
        memberCount: 0,
        workspaceId: workspaces[0].id,
      },
    }),
    prisma.channel.create({
      data: {
        name: 'team-leads',
        description: 'Private channel for team leads',
        isPrivate: true,
        memberCount: 0,
        workspaceId: workspaces[0].id,
      },
    }),
    // Project Alpha workspace channels
    prisma.channel.create({
      data: {
        name: 'development',
        description: 'Technical discussions for developers',
        isPrivate: false,
        memberCount: 0,
        workspaceId: workspaces[1].id,
      },
    }),
    prisma.channel.create({
      data: {
        name: 'planning',
        description: 'Project planning and timelines',
        isPrivate: false,
        memberCount: 0,
        workspaceId: workspaces[1].id,
      },
    }),
  ])

  console.log(`Created ${channels.length} channels`)
  return channels
}

async function addUsersToChannels(users: User[], channels: Channel[]): Promise<void> {
  const channelMembers = []

  // Add all users to general channel in main workspace
  for (const user of users) {
    channelMembers.push(
      await prisma.channelMember.create({
        data: {
          userId: user.id,
          channelId: channels[0].id, // general channel
        },
      })
    )
  }

  // Add all users to random channel in main workspace
  for (const user of users) {
    channelMembers.push(
      await prisma.channelMember.create({
        data: {
          userId: user.id,
          channelId: channels[1].id, // random channel
        },
      })
    )
  }

  // Add first 2 users to team-leads channel in main workspace
  for (let i = 0; i < 2; i++) {
    channelMembers.push(
      await prisma.channelMember.create({
        data: {
          userId: users[i].id,
          channelId: channels[2].id, // team-leads channel
        },
      })
    )
  }

  // Add first 3 users to development channel in Project Alpha workspace
  for (let i = 0; i < 3; i++) {
    channelMembers.push(
      await prisma.channelMember.create({
        data: {
          userId: users[i].id,
          channelId: channels[3].id, // development channel
        },
      })
    )
  }

  // Add first 3 users to planning channel in Project Alpha workspace
  for (let i = 0; i < 3; i++) {
    channelMembers.push(
      await prisma.channelMember.create({
        data: {
          userId: users[i].id,
          channelId: channels[4].id, // planning channel
        },
      })
    )
  }

  // Update member counts
  for (const channel of channels) {
    const count = await prisma.channelMember.count({
      where: {
        channelId: channel.id,
      },
    })

    await prisma.channel.update({
      where: {
        id: channel.id,
      },
      data: {
        memberCount: count,
      },
    })
  }

  console.log(`Added ${channelMembers.length} channel memberships`)
}

async function createDirectConversations(users: User[]): Promise<DirectConversation[]> {
  const conversations: DirectConversation[] = []

  // 1:1 conversations between users
  for (let i = 0; i < users.length; i++) {
    for (let j = i + 1; j < users.length; j++) {
      const conversation = await prisma.directConversation.create({
        data: {
          isGroup: false,
          name: [users[i].name, users[j].name].join(', '),
          creatorId: users[i].id,
          members: {
            create: [
              {
                userId: users[i].id,
              },
              {
                userId: users[j].id,
              },
            ],
          },
        },
      })
      conversations.push(conversation)
    }
  }

  // One group conversation with all users
  const groupConversation = await prisma.directConversation.create({
    data: {
      isGroup: true,
      name: 'Team Chat',
      creatorId: users[0].id,
      members: {
        create: users.map((user) => ({
          userId: user.id,
        })),
      },
    },
  })
  conversations.push(groupConversation)

  console.log(`Created ${conversations.length} direct conversations`)
  return conversations
}

async function createChannelMessages(users: User[], channels: Channel[]): Promise<void> {
  const messages: Message[] = []
  const now = new Date()

  // Add messages to general channel
  for (let i = 0; i < 10; i++) {
    const sender = users[i % users.length]
    const messageTime = new Date(now.getTime() - (10 - i) * 3600000) // Hours ago

    const message = await prisma.message.create({
      data: {
        content: encryptAES(`This is message #${i + 1} in general channel`),
        senderId: sender.id,
        channelId: channels[0].id,
        createdAt: messageTime,
        updatedAt: messageTime,
      },
    })
    messages.push(message)

    // Add some reactions to general channel messages
    if (i % 3 === 0) {
      await prisma.reaction.create({
        data: {
          emoji: '👍',
          userId: users[(i + 1) % users.length].id,
          messageId: message.id,
        },
      })
    }

    // Create thread replies for some messages
    if (i % 4 === 0) {
      const threadMessage = await prisma.message.create({
        data: {
          content: encryptAES(`This is a thread reply to message #${i + 1}`),
          senderId: users[(i + 2) % users.length].id,
          channelId: channels[0].id,
          parentMessageId: message.id,
          createdAt: new Date(messageTime.getTime() + 1800000), // 30 mins later
          updatedAt: new Date(messageTime.getTime() + 1800000),
        },
      })
      messages.push(threadMessage)
    }
  }

  // Add messages to development channel
  for (let i = 0; i < 5; i++) {
    const sender = users[i % 3] // Only first 3 users are in this channel
    const messageTime = new Date(now.getTime() - (5 - i) * 3600000)

    const message = await prisma.message.create({
      data: {
        content: encryptAES(`Development update #${i + 1}: Working on new features`),
        senderId: sender.id,
        channelId: channels[3].id, // Now referring to development channel in Project Alpha workspace
        createdAt: messageTime,
        updatedAt: messageTime,
      },
    })
    messages.push(message)
  }

  // Add messages to team-leads channel
  for (let i = 0; i < 3; i++) {
    const sender = users[i % 2] // Only first 2 users are in this channel
    const messageTime = new Date(now.getTime() - (3 - i) * 3600000)

    const message = await prisma.message.create({
      data: {
        content: encryptAES(`Team lead announcement #${i + 1}`),
        senderId: sender.id,
        channelId: channels[2].id, // Now referring to team-leads channel in Main workspace
        createdAt: messageTime,
        updatedAt: messageTime,
      },
    })
    messages.push(message)
  }

  // Add messages to planning channel in Project Alpha
  for (let i = 0; i < 4; i++) {
    const sender = users[i % 3] // Only first 3 users are in this channel
    const messageTime = new Date(now.getTime() - (4 - i) * 3600000)

    const message = await prisma.message.create({
      data: {
        content: encryptAES(
          `Planning item #${i + 1}: ${i % 2 === 0 ? 'Timeline updates' : 'Resource allocation'}`
        ),
        senderId: sender.id,
        channelId: channels[4].id, // Planning channel in Project Alpha
        createdAt: messageTime,
        updatedAt: messageTime,
      },
    })
    messages.push(message)
  }

  // Add attachments to some messages
  const randomMessage = messages[Math.floor(Math.random() * messages.length)]
  await prisma.attachment.create({
    data: {
      messageId: randomMessage.id,
      url: 'https://example.com/files/document.pdf',
      contentType: 'application/pdf',
      fileName: 'project_plan.pdf',
      size: 1024 * 1024, // 1MB
    },
  })

  // Update lastMessage fields for each channel
  for (const channel of channels) {
    const lastMessage = await prisma.message.findFirst({
      where: {
        channelId: channel.id,
        parentMessageId: null, // Only top-level messages
      },
      orderBy: {
        createdAt: 'desc',
      },
    })

    if (lastMessage) {
      await prisma.channel.update({
        where: {
          id: channel.id,
        },
        data: {
          lastMessageId: lastMessage.id,
          lastMessageAt: lastMessage.createdAt,
          lastMessagePreview: lastMessage.content.substring(0, 100),
          lastMessageSenderId: lastMessage.senderId,
        },
      })
    }
  }

  console.log(`Created ${messages.length} channel messages`)
}

type ConversationMember = {
  userId: string
  conversationId: string
  user: User
}

async function createDirectMessages(
  users: User[],
  conversations: DirectConversation[]
): Promise<void> {
  const messages: Message[] = []
  const now = new Date()

  // Add messages to each direct conversation (1:1)
  for (let i = 0; i < conversations.length - 1; i++) {
    // Exclude group conversation
    const conversation = conversations[i]

    // Get the members of this conversation
    const members = (await prisma.directConversationMember.findMany({
      where: {
        conversationId: conversation.id,
      },
      include: {
        user: true,
      },
    })) as unknown as ConversationMember[]

    // Create messages between these two users
    for (let j = 0; j < 5; j++) {
      const sender = members[j % 2].user
      const messageTime = new Date(now.getTime() - (5 - j) * 1800000) // 30 mins intervals

      const message = await prisma.message.create({
        data: {
          content: encryptAES(`Direct message ${j + 1} from ${sender.name}`),
          senderId: sender.id,
          directConversationId: conversation.id,
          createdAt: messageTime,
          updatedAt: messageTime,
        },
      })
      messages.push(message)

      // Create message status entries for DM
      const recipient = members.find((m) => m.user.id !== sender.id)
      if (recipient) {
        await prisma.directMessageStatus.create({
          data: {
            messageId: message.id,
            userId: recipient.user.id,
            deliveredAt: new Date(messageTime.getTime() + 5000), // 5 seconds later
            readAt: j < 3 ? new Date(messageTime.getTime() + 60000) : null, // First 3 are read, rest are unread
          },
        })
      }
    }
  }

  // Add messages to group conversation (last in the list)
  const groupConversation = conversations[conversations.length - 1]

  for (let i = 0; i < 8; i++) {
    const sender = users[i % users.length]
    const messageTime = new Date(now.getTime() - (8 - i) * 3600000)

    const message = await prisma.message.create({
      data: {
        content: encryptAES(`Group message ${i + 1} from ${sender.name}`),
        senderId: sender.id,
        directConversationId: groupConversation.id,
        createdAt: messageTime,
        updatedAt: messageTime,
      },
    })
    messages.push(message)

    // Create message status entries for group message
    for (const user of users) {
      if (user.id !== sender.id) {
        await prisma.directMessageStatus.create({
          data: {
            messageId: message.id,
            userId: user.id,
            deliveredAt: new Date(messageTime.getTime() + 5000), // 5 seconds later
            readAt: i < 5 ? new Date(messageTime.getTime() + 300000) : null, // First 5 are read by everyone
          },
        })
      }
    }
  }

  // Update lastMessage fields for each conversation
  for (const conversation of conversations) {
    const lastMessage = await prisma.message.findFirst({
      where: {
        directConversationId: conversation.id,
      },
      orderBy: {
        createdAt: 'desc',
      },
    })

    if (lastMessage) {
      await prisma.directConversation.update({
        where: {
          id: conversation.id,
        },
        data: {
          lastMessageId: lastMessage.id,
          lastMessageAt: lastMessage.createdAt,
          lastMessagePreview: lastMessage.content.substring(0, 100),
          lastMessageSenderId: lastMessage.senderId,
        },
      })
    }
  }

  console.log(`Created ${messages.length} direct messages`)
}

async function updateMessageStatuses(): Promise<void> {
  // Get all direct conversations to process their message statuses
  const conversations = await prisma.directConversation.findMany({
    include: {
      members: {
        include: {
          user: true,
        },
      },
      messages: {
        orderBy: {
          createdAt: 'asc',
        },
        include: {
          directStatuses: true,
        },
      },
    },
  })

  for (const conversation of conversations) {
    // For each member in the conversation
    for (const member of conversation.members) {
      const userId = member.userId

      // Get messages in this conversation that this user didn't send
      const messagesFromOthers = conversation.messages.filter(
        (message) => message.senderId !== userId
      )

      if (messagesFromOthers.length === 0) continue

      // Get the last message in the conversation (regardless of sender)
      const lastMessage = conversation.messages[conversation.messages.length - 1]

      // If the last message is from this user, mark all previous messages as read
      if (lastMessage.senderId === userId) {
        for (const message of messagesFromOthers) {
          const status = message.directStatuses.find((s) => s.userId === userId)
          if (status && !status.readAt) {
            await prisma.directMessageStatus.update({
              where: {
                id: status.id,
              },
              data: {
                readAt: new Date(message.createdAt.getTime() + 60000), // Read 1 minute after creation
              },
            })
          }
        }
      } else {
        // Otherwise, apply the 3 read + 1-2 unread logic
        for (let i = 0; i < messagesFromOthers.length; i++) {
          const message = messagesFromOthers[i]
          const status = message.directStatuses.find((s) => s.userId === userId)

          if (status && !status.readAt) {
            // Read first 3 messages, leave last 1-2 unread
            const shouldBeRead =
              i < Math.max(0, messagesFromOthers.length - Math.floor(Math.random() * 2 + 1))

            if (shouldBeRead) {
              await prisma.directMessageStatus.update({
                where: {
                  id: status.id,
                },
                data: {
                  readAt: new Date(message.createdAt.getTime() + 60000), // Read 1 minute after creation
                },
              })
            }
          }
        }
      }
    }
  }
}
